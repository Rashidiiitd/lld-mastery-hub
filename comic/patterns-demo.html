struct PricingStrategy {
  virtual double price(double base) const = 0;
  virtual ~PricingStrategy() = default;
};
struct Regular : PricingStrategy {
  double price(double b) const override { return b; }
};
struct Sale : PricingStrategy {
  double price(double b) const override { return b * 0.8; }
};
class Cart {
  std::unique_ptr<PricingStrategy> s;
public:
  explicit Cart(std::unique_ptr<PricingStrategy> p) : s(std::move(p)) {}
  double total(double base) const { return s->price(base); }
};
  • Easy to extend and test; no more growing if/else chain.
  • Can add many small types if overused.
  • The caller must still know which strategy to pick.

using Callback = std::function<void(const Event&)>;
class Subject {
  std::map<int, Callback> subs;
  int next = 0;
public:
  int subscribe(Callback cb) { subs[next] = std::move(cb); return next++; }
  void unsubscribe(int id) { subs.erase(id); }
  void publish(const Event& e) {
    auto copy = subs;  // callbacks may unsubscribe while we loop
    for (auto& kv : copy) kv.second(e);
  }
};
  • Decouples components: the subject never names its consumers.
  • Event ordering, lifetime, and re-entrancy need care.
  • Dangling callbacks and thread-safety bugs are easy to introduce.

LoggingNotifier wraps another Notifier: same interface in, same interface out

struct Notifier {
  virtual void send(std::string_view msg) = 0;
  virtual ~Notifier() = default;
};
struct EmailNotifier : Notifier {
  void send(std::string_view msg) override { /* send email */ }
};
class LoggingNotifier : public Notifier {
  std::unique_ptr<Notifier> inner;
public:
  explicit LoggingNotifier(std::unique_ptr<Notifier> n) : inner(std::move(n)) {}
  void send(std::string_view msg) override {
    std::cout << "send: " << msg << '\n';
    inner->send(msg);
  }
};
  • Mix optional features at runtime without a subclass explosion.
  • Debugging nested wrappers may take extra effort.
  • Many small objects to create and keep track of.

struct Button {
  virtual void draw() const = 0;
  virtual ~Button() = default;
};
struct WinButton : Button { void draw() const override { /* ... */ } };
struct MacButton : Button { void draw() const override { /* ... */ } };

struct Dialog {
  virtual std::unique_ptr<Button> make() const = 0;  // factory method
  void render() const { make()->draw(); }
  virtual ~Dialog() = default;
};
struct WinDialog : Dialog {
  std::unique_ptr<Button> make() const override { return std::make_unique<WinButton>(); }
};
  • Client code depends on the interface, not on concrete classes.
  • Add a new product without touching existing client code.
  • One extra subclass per product variant.
  • Often a plain function or a map of creators is enough.
